{{- /* Overrides the hugo-book theme's link hook so that external links open in a new tab. External: another site, or this domain outside the Hugo docs (/docs/ and the home page), e.g. the generated API docs at /pyDataPipeline/ or /DataPipeline.jl/. */ -}} {{- $destination := .Destination -}} {{- if .Page.Site.Params.BookPortableLinks -}} {{- $destination = partial "docs/links/portable-link" . -}} {{- end -}} {{- $url := urls.Parse .Destination -}} {{- $siteHost := strings.TrimPrefix "www." (urls.Parse site.BaseURL).Host -}} {{- $external := false -}} {{- if in (slice "http" "https") $url.Scheme -}} {{- if ne (strings.TrimPrefix "www." $url.Host) $siteHost -}} {{- $external = true -}} {{- else if not (or (in (slice "" "/") $url.Path) (strings.HasPrefix $url.Path "/docs/")) -}} {{- $external = true -}} {{- end -}} {{- else if and (eq $url.Scheme "") (eq $url.Host "") (strings.HasPrefix $url.Path "/") -}} {{- if not (or (eq $url.Path "/") (strings.HasPrefix $url.Path "/docs/")) -}} {{- $external = true -}} {{- end -}} {{- end -}} {{ .Text | safeHTML }} {{- /**/ -}}
